cmake_minimum_required(VERSION 3.21)
project(omarchy-fingerprint LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DBUS REQUIRED IMPORTED_TARGET dbus-1)
find_package(Qt6 REQUIRED COMPONENTS Quick QuickControls2 DBus Concurrent Sql Network)
add_library(fingerprint-core STATIC reader.cpp names.cpp discovery.cpp backend.cpp appearance.cpp)
set(FINGERPRINT_PAM_HELPER "/usr/share/omarchy/default/fingerprint/sudo_pam.py" CACHE FILEPATH "Installed root-owned sudo helper")
target_compile_definitions(fingerprint-core PRIVATE FINGERPRINT_PAM_HELPER="${FINGERPRINT_PAM_HELPER}")
target_link_libraries(fingerprint-core PUBLIC Qt6::Quick Qt6::DBus Qt6::Concurrent Qt6::Sql Qt6::Network PkgConfig::DBUS)
add_executable(fingerprint-setup main.cpp)
qt_add_resources(fingerprint-setup interface PREFIX "/" FILES Main.qml Fingerprint.qml)
qt_add_resources(fingerprint-setup messages PREFIX "/i18n" BASE "../../default/i18n/en" FILES "../../default/i18n/en/cli.json")
target_link_libraries(fingerprint-setup PRIVATE fingerprint-core Qt6::QuickControls2)
install(TARGETS fingerprint-setup RUNTIME DESTINATION lib/omarchy)
include(CTest)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
