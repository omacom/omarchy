cmake_minimum_required(VERSION 3.21)
project(OmarchyAmigaRuntime VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client)
pkg_check_modules(PULSE REQUIRED IMPORTED_TARGET libpulse)
pkg_get_variable(PROTOCOLS wayland-protocols pkgdatadir)
find_program(SCANNER wayland-scanner REQUIRED)
set(PROTOCOL ${PROTOCOLS}/unstable/relative-pointer/relative-pointer-unstable-v1.xml)
add_custom_command(OUTPUT relative-pointer.h COMMAND ${SCANNER} client-header ${PROTOCOL} relative-pointer.h)
add_custom_command(OUTPUT relative-pointer.c COMMAND ${SCANNER} private-code ${PROTOCOL} relative-pointer.c)
add_library(amigainput SHARED plugin.cpp relative-pointer.c relative-pointer.h)
target_include_directories(amigainput PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(amigainput PRIVATE Qt6::Core Qt6::Gui Qt6::Qml PkgConfig::WAYLAND)
add_library(amiga-pulse SHARED pulse-start.c)
target_link_libraries(amiga-pulse PRIVATE PkgConfig::PULSE dl)
set(RUNTIME lib/omarchy-amiga-runtime)
pkg_check_modules(SDL REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(GL REQUIRED IMPORTED_TARGET gl)
add_executable(gl-probe gl-probe.c)
target_link_libraries(gl-probe PRIVATE PkgConfig::SDL PkgConfig::GL)
install(TARGETS gl-probe RUNTIME DESTINATION ${RUNTIME}/bin)
install(TARGETS amigainput LIBRARY DESTINATION ${RUNTIME}/guard/AmigaInput)
install(TARGETS amiga-pulse LIBRARY DESTINATION ${RUNTIME}/audio)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/qmldir "module AmigaInput\nplugin amigainput\n")
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/qmldir DESTINATION ${RUNTIME}/guard/AmigaInput)
install(FILES guard/Guard.qml DESTINATION ${RUNTIME}/guard)
install(FILES ../../shell/plugins/services/idle/state.py ../../shell/plugins/services/idle/pack.py
  ../../shell/plugins/services/idle/renderer.py ../../shell/plugins/services/idle/runtime.py ../../shell/plugins/services/idle/history.py ../../shell/plugins/services/idle/amiga.py
  ../../shell/plugins/services/idle/audio.py
  ../../shell/plugins/services/idle/i18n.json DESTINATION ${RUNTIME}/controller)
# The private OpenAL is deliberately linked to Pulse; dlopen bypasses our hook.
if(NOT OPENAL_SOURCE)
  message(FATAL_ERROR "Set OPENAL_SOURCE to the pinned OpenAL Soft source checkout")
endif()
execute_process(COMMAND git -C ${OPENAL_SOURCE} rev-parse HEAD OUTPUT_VARIABLE OPENAL_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT OPENAL_COMMIT STREQUAL "b2c48f7718ef3fcf67921a8b6534c4914e328970")
  message(FATAL_ERROR "OpenAL Soft source must be pinned to 1.25.2")
endif()
execute_process(COMMAND git -C ${OPENAL_SOURCE} diff --quiet ${OPENAL_COMMIT} -- RESULT_VARIABLE OPENAL_DIRTY)
if(NOT OPENAL_DIRTY EQUAL 0)
  message(FATAL_ERROR "OpenAL Soft tracked sources differ from the pinned commit")
endif()
set(ALSOFT_DLOPEN OFF CACHE BOOL "" FORCE)
set(ALSOFT_UTILS OFF CACHE BOOL "" FORCE)
set(ALSOFT_EXAMPLES OFF CACHE BOOL "" FORCE)
foreach(backend PIPEWIRE ALSA JACK OSS SNDIO PORTAUDIO)
  set(ALSOFT_BACKEND_${backend} OFF CACHE BOOL "" FORCE)
endforeach()
set(ALSOFT_REQUIRE_PULSEAUDIO ON CACHE BOOL "" FORCE)
set(ALSOFT_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(${OPENAL_SOURCE} openal EXCLUDE_FROM_ALL)
install(FILES $<TARGET_FILE:OpenAL> DESTINATION ${RUNTIME}/audio RENAME libopenal.so.1)
add_dependencies(amiga-pulse OpenAL)
install(FILES ${OPENAL_SOURCE}/COPYING DESTINATION share/licenses/omarchy-amiga-runtime RENAME OpenAL-COPYING)
install(FILES ${OPENAL_SOURCE}/LICENSE-pffft DESTINATION share/licenses/omarchy-amiga-runtime)
install(FILES ${OPENAL_SOURCE}/gsl/LICENSE DESTINATION share/licenses/omarchy-amiga-runtime RENAME GSL-LICENSE)
install(FILES ${OPENAL_SOURCE}/fmt-11.2.0/LICENSE DESTINATION share/licenses/omarchy-amiga-runtime RENAME fmt-LICENSE)
install(FILES ../../LICENSE DESTINATION share/licenses/omarchy-amiga-runtime RENAME Omarchy-LICENSE)
